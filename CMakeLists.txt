cmake_minimum_required(VERSION 3.16)
project(SmartRide VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SANITIZER "" CACHE STRING "address or thread")
if(SANITIZER)
  add_compile_options(-fsanitize=${SANITIZER} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${SANITIZER})
endif()
find_package(Threads REQUIRED)
add_library(smartride src/graph.cpp src/routing.cpp src/matching.cpp)
target_include_directories(smartride PUBLIC include)
target_link_libraries(smartride PUBLIC Threads::Threads)
target_compile_options(smartride PRIVATE -Wall -Wextra -Wpedantic)
add_executable(smartride_cli src/main.cpp)
target_link_libraries(smartride_cli PRIVATE smartride)
enable_testing()
add_executable(smartride_tests tests/test.cpp)
target_link_libraries(smartride_tests PRIVATE smartride)
add_test(NAME correctness COMMAND smartride_tests)
add_executable(routing_tests tests/routing_test.cpp)
target_link_libraries(routing_tests PRIVATE smartride)
add_test(NAME routing COMMAND routing_tests)
add_executable(matching_tests tests/matching_test.cpp)
target_link_libraries(matching_tests PRIVATE smartride)
add_test(NAME matching COMMAND matching_tests)
set_tests_properties(correctness routing matching PROPERTIES TIMEOUT 120)
add_executable(matching_bench benchmarks/matching_bench.cpp)
target_link_libraries(matching_bench PRIVATE smartride)
add_executable(batch_matching_bench benchmarks/batch_matching_bench.cpp)
target_link_libraries(batch_matching_bench PRIVATE smartride)
add_executable(hungarian_matching_bench benchmarks/hungarian_matching_bench.cpp)
target_link_libraries(hungarian_matching_bench PRIVATE smartride)
